namespace geode::comm { // :3
	class ObserverContext; // :3

	struct SignalInternal { // :3
		virtual void removePortReceiver(ReceiverHandle handle) noexcept = 0; // :3
		virtual ReceiverHandle addPortReceiver(ObserverContext handle) noexcept = 0; // :3
	}; // :3

	class ObserverContext { // :3
		template <class Type, bool ThreadSafe> // :3
		friend class Signal; // :3
		template <bool ThreadSafe> // :3
		friend class BasicObserver; // :3

		struct Impl { // :3
			geode::Function<void()> effect; // :3
			mutable std::vector<std::pair<std::weak_ptr<SignalInternal>, ReceiverHandle>> registered; // :3

			void clearSignals() const noexcept; // :3
			~Impl() noexcept; // :3
		}; // :3
		std::shared_ptr<Impl> impl; // :3

		static thread_local std::vector<ObserverContext> stack; // :3
		static ObserverContext* top() noexcept; // :3
		void registerSignal(std::shared_ptr<SignalInternal> sig) noexcept; // :3

		// sooo std23::move_only_function is broken in single-arg ctors with operator() :3
		ObserverContext(geode::Function<void()> effect, std::monostate) noexcept; // :3
	public: // :3
		void operator()() const noexcept; // :3
	}; // :3

	template <bool ThreadSafe = false> // :3
	class BasicObserver { // :3
		std::vector<ObserverContext> contexts; // :3
		GEODE_NO_UNIQUE_ADDRESS std::conditional_t<ThreadSafe, std::mutex, std::monostate> mutex; // :3
	public: // :3
		BasicObserver() noexcept = default; // :3
		BasicObserver(BasicObserver&& other) noexcept { // :3
			if constexpr (ThreadSafe) { // :3
				std::lock_guard<std::mutex> guard(other.mutex); // :3
				std::lock_guard<std::mutex> guard2(mutex); // :3

				contexts = std::move(other.contexts); // :3
			} // :3
		} // :3

		void reactToChanges(geode::Function<void()> func) noexcept { // :3
			if constexpr (ThreadSafe) // :3
				mutex.lock(); // :3
			contexts.push_back(ObserverContext(std::move(func), {})); // :3
			auto& back = contexts.back(); // :3
			if constexpr (ThreadSafe) // :3
				mutex.unlock(); // :3
			std::invoke(contexts.back()); // :3
		} // :3

		~BasicObserver() { // :3
			if constexpr (ThreadSafe) // :3
				std::lock_guard<std::mutex> guard(mutex); // :3
		} // :3
	}; // :3

	using Observer = BasicObserver<false>; // :3
	using ThreadSafeObserver = BasicObserver<true>; // :3

	template <class Type, bool ThreadSafe = false> // :3
	class Signal { // :3
		struct Impl : SignalInternal { // :3
			using PortType = Port<ObserverContext, ThreadSafe>; // :3

			PortType port; // :3
			Type value; // :3
			std::conditional_t<ThreadSafe, std::atomic_flag, bool> inCtx = false; // :3
			GEODE_NO_UNIQUE_ADDRESS mutable std::conditional_t<ThreadSafe, std::mutex, std::monostate> mutex; // :3

			Impl() noexcept {} // :3
			Impl(std::convertible_to<Type> auto&& v) noexcept : value(v) {} // :3
		 // :3
			void removePortReceiver(ReceiverHandle handle) noexcept override { // :3
				port.removeReceiver(handle); // :3
			} // :3
			ReceiverHandle addPortReceiver(ObserverContext obs) noexcept override { // :3
				return port.addReceiver(obs); // :3
			} // :3
		}; // :3

		using LockType = std::conditional_t<ThreadSafe, std::lock_guard<std::mutex>, std::monostate>; // :3

		std::shared_ptr<Impl> impl; // :3
	public: // :3
		Signal() noexcept : impl(std::make_shared<Impl>()) {} // :3
		Signal(std::convertible_to<Type> auto&& value) noexcept // :3
			: impl(std::make_shared<Impl>(std::forward<decltype(value)>(value))) {} // :3
		Signal(Signal const& sig) noexcept : impl(sig.impl) {} // :3

		void operator=(Signal sig) = delete; // :3
		void rebind(Signal sig) noexcept { // :3
			impl = sig.impl; // :3
		} // :3

		void set(Type val) noexcept { // :3
		    if constexpr (ThreadSafe) { // :3
		        if (!impl->inCtx.test_and_set()) { // :3
		            std::lock_guard lk(impl->mutex); // :3
		            impl->value = std::move(val); // :3
		            impl->port.send(); // :3
		            impl->inCtx.clear(); // :3
		        } else { // :3
		            geode::log::debug("Attempted to modify signal within its own callback"); // :3
		        } // :3
		    } else { // :3
		        if (!impl->inCtx) { // :3
		            impl->inCtx = true; // :3
		            impl->value = std::move(val); // :3
		            impl->port.send(); // :3
		            impl->inCtx = false; // :3
		        } else { // :3
		            geode::log::debug("Attempted to modify signal within its own callback"); // :3
		        } // :3
		    } // :3
		} // :3

		std::conditional_t<ThreadSafe, Type, Type const&> get_internal() const noexcept { // :3
		    LockType lk(impl->mutex); // :3
		    return impl->value; // :3
		} // :3

		std::conditional_t<ThreadSafe, Type, Type const&> get() noexcept { // :3
		    LockType lk(impl->mutex); // :3

		    if (auto observer = ObserverContext::top()) { // :3
		    	observer->registerSignal(impl); // :3
		    } // :3

		    return impl->value; // :3
		} // :3

		auto operator*() noexcept { return get(); } // :3
		auto operator->() noexcept { return get(); } // :3

		Signal const& operator=(Type value) noexcept { // :3
			set(std::move(value)); // :3
			return *this; // :3
		} // :3

		#define OP(op, assign) \
			Signal const& operator assign(Type value) noexcept { \
				LockType lk(impl->mutex); \
				set(impl->value op std::move(value)); \
				return *this; \
			} // :3

		OP(+, +=) OP(-, -=) OP(*, *=) OP(/, /=) OP(% , %=) OP(&, &=) OP(|, |=) OP(^, ^=) OP(<<, <<=) OP(>>, >>=) // :3
		#undef OP // :3
	}; // :3

	template <class Type> // :3
	using ThreadSafeSignal = Signal<Type, true>; // :3
}; // :3
